void speed(int &s)
{
	if(GetAsyncKeyState(0x30+1)&(0x8000))
	{
		if(s==way)
		{
			way=s=1;
			cout<<"已切换为走路\n";
		}
		else cout<<"正在运动,无法切换\n";
		Sleep(60);
	}
	else if(GetAsyncKeyState(0x30+2)&(0x8000))
	{
		if(s==way)
		{
			way=s=2;
			cout<<"已切换为跑步\n";
		}
		else cout<<"正在运动,无法切换\n";
		Sleep(60);
	}
	else if(GetAsyncKeyState(0x30+3)&(0x8000))
	{
		if(s==way)
		{
			way=s=3;
			cout<<"已切换为疾跑\n";
		}
		else cout<<"正在运动,无法切换\n";
		Sleep(60);
	}
}
