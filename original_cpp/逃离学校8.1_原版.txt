void hide() {
    HANDLE hOut = GetStdHandle(STD_OUTPUT_HANDLE);
    CONSOLE_CURSOR_INFO cursorInfo;
    GetConsoleCursorInfo(hOut, &cursorInfo);

}

void speed(int &s)
{
	if(GetAsyncKeyState(0x30+1)&(0x8000))
	{
		if(s==way)
		{
			way=s=1;
			cout<<"已切换为走路\n";
			start_end.push_back("	cout<<\"已切换为走路\\n\";\n");
		}
		else cout<<"正在运动,无法切换\n";
		Sleep(60);
	}
